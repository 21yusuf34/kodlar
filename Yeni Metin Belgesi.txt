using UnityEngine;

[RequireComponent(typeof(Rigidbody2D))]
[RequireComponent(typeof(SpriteRenderer))]
[RequireComponent(typeof(Animator))]
public class PlayerController : MonoBehaviour
{
    [Header("Hareket Ayarları")]
    public float moveSpeed = 5f;
    public float jumpForce = 12f;

    [Header("Zemin Kontrolü")]
    public Transform groundCheck;
    public float groundCheckRadius = 0.2f;
    public LayerMask groundLayer;

    private Rigidbody2D rb;
    private SpriteRenderer spriteRenderer;
    private Animator animator;

    private float moveInput;
    private bool isGrounded;

    void Start()
    {
        rb = GetComponent<Rigidbody2D>();
        spriteRenderer = GetComponent<SpriteRenderer>();
        animator = GetComponent<Animator>();
    }

    void Update()
    {
        // Yatay hareket girdisi (-1, 0, 1)
        moveInput = Input.GetAxisRaw("Horizontal");

        // Yerde mi kontrolü
        isGrounded = Physics2D.OverlapCircle(
            groundCheck.position,
            groundCheckRadius,
            groundLayer
        );

        // Zıplama girdisi
        if (Input.GetKeyDown(KeyCode.Space) && isGrounded)
        {
            rb.linearVelocity = new Vector2(
                rb.linearVelocity.x,
                jumpForce
            );
        }

        // Karakterin yönünü çevirme (Sağ / Sol)
        if (moveInput > 0)
        {
            spriteRenderer.flipX = false; // Sağa bak
        }
        else if (moveInput < 0)
        {
            spriteRenderer.flipX = true;  // Sola bak
        }

        // --- ÇÖZÜM KISMI ---
        // moveInput 0'dan farklıysa (aşırı küçük kaymaları önlemek için Mathf.Abs kullanıyoruz) yürüyor kabul et.
        bool isMoving = Mathf.Abs(moveInput) > 0.01f;

        // Animator parametrelerini güncelleme
        animator.SetBool("isWalking", isMoving);
        animator.SetBool("isGrounded", isGrounded);
    }

    void FixedUpdate()
    {
        // Karakterin yatay hareketi
        rb.linearVelocity = new Vector2(
            moveInput * moveSpeed,
            rb.linearVelocity.y
        );
    }
}
